<script>
(function () {
  var STORAGE_KEY = 'momentum-habits-v1';
  var DAYS = 30;

  function todayStr() {
    var d = new Date();
    return d.getFullYear() + '-' + String(d.getMonth()+1).padStart(2,'0') + '-' + String(d.getDate()).padStart(2,'0');
  }
  function dateStr(offset) {
    var d = new Date();
    d.setDate(d.getDate() - offset);
    return d.getFullYear() + '-' + String(d.getMonth()+1).padStart(2,'0') + '-' + String(d.getDate()).padStart(2,'0');
  }

  function load() {
    try {
      var raw = localStorage.getItem(STORAGE_KEY);
      return raw ? JSON.parse(raw) : [];
    } catch (e) { return []; }
  }
  function save(habits) {
    try { localStorage.setItem(STORAGE_KEY, JSON.stringify(habits)); } catch (e) {}
  }

  var habits = load();

  function currentStreak(h) {
    var streak = 0;
    for (var i = 0; i < 400; i++) {
      var ds = dateStr(i);
      if (h.log && h.log[ds]) { streak++; }
      else {
        if (i === 0) continue;
        break;
      }
    }
    return streak;
  }

  function render() {
    var list = document.getElementById('habitList');
    var empty = document.getElementById('emptyState');
    list.innerHTML = '';

    if (habits.length === 0) {
      empty.style.display = 'block';
    } else {
      empty.style.display = 'none';
    }

    var doneToday = 0;
    var longest = 0;
    var today = todayStr();

    habits.forEach(function (h, idx) {
      if (h.log && h.log[today]) doneToday++;
      var streak = currentStreak(h);
      if (streak > longest) longest = streak;

      var card = document.createElement('div');
      card.className = 'habit';

      var top = document.createElement('div');
      top.className = 'habit-top';

      var name = document.createElement('div');
      name.className = 'habit-name';
      name.textContent = h.name;

      var meta = document.createElement('div');
      meta.className = 'habit-meta';

      var streakBadge = document.createElement('span');
      streakBadge.className = 'streak';
      streakBadge.innerHTML = '&#9679; ' + streak + (streak === 1 ? ' day' : ' days');

      var doneBtn = document.createElement('button');
      doneBtn.type = 'button';
      var isDoneToday = !!(h.log && h.log[today]);
      doneBtn.className = 'today-btn' + (isDoneToday ? ' done' : '');
      doneBtn.textContent = isDoneToday ? 'Done today' : 'Mark today';
      doneBtn.addEventListener('click', function () {
        h.log = h.log || {};
        if (h.log[today]) { delete h.log[today]; }
        else { h.log[today] = true; }
        save(habits);
        render();
      });

      var delBtn = document.createElement('button');
      delBtn.type = 'button';
      delBtn.className = 'del-btn';
      delBtn.setAttribute('aria-label', 'Delete habit ' + h.name);
      delBtn.textContent = 'Remove';
      delBtn.addEventListener('click', function () {
        habits.splice(idx, 1);
        save(habits);
        render();
      });

      meta.appendChild(streakBadge);
      meta.appendChild(doneBtn);
      meta.appendChild(delBtn);
      top.appendChild(name);
      top.appendChild(meta);

      var grid = document.createElement('div');
      grid.className = 'grid30';
      for (var i = DAYS - 1; i >= 0; i--) {
        var cell = document.createElement('div');
        var ds = dateStr(i);
        cell.className = 'cell' + ((h.log && h.log[ds]) ? ' on' : '');
        cell.title = ds;
        grid.appendChild(cell);
      }

      card.appendChild(top);
      card.appendChild(grid);
      list.appendChild(card);
    });

    document.getElementById('statHabits').textContent = habits.length;
    document.getElementById('statToday').textContent = doneToday;
    document.getElementById('statStreak').textContent = longest;
  }

  document.getElementById('addBtn').addEventListener('click', addHabit);
  document.getElementById('newHabit').addEventListener('keydown', function (e) {
    if (e.key === 'Enter') addHabit();
  });

  function addHabit() {
    var input = document.getElementById('newHabit');
    var errorEl = document.getElementById('errorMsg');
    var val = input.value.trim();
    if (!val) {
      errorEl.classList.add('show');
      input.focus();
      return;
    }
    errorEl.classList.remove('show');
    habits.push({ name: val, log: {} });
    save(habits);
    input.value = '';
    render();
  }

  document.getElementById('newHabit').addEventListener('input', function () {
    document.getElementById('errorMsg').classList.remove('show');
  });

  var toggle = document.getElementById('themeToggle');
  var root = document.documentElement;
  function applyTheme(t) {
    if (t) root.setAttribute('data-theme', t);
    toggle.textContent = (root.getAttribute('data-theme') === 'dark') ? 'Dark' : 'Light';
  }
  try {
    var savedTheme = localStorage.getItem('momentum-theme');
    if (savedTheme) applyTheme(savedTheme);
  } catch (e) {}
  toggle.addEventListener('click', function () {
    var next = root.getAttribute('data-theme') === 'dark' ? 'light' : 'dark';
    applyTheme(next);
    try { localStorage.setItem('momentum-theme', next); } catch (e) {}
  });

  render();
})();
</script>
